#!/usr/bin/env bash
# Set supported laptop batteries to stop charging at 80%.
# Run with sudo: sudo bash backup.txt

set -euo pipefail

found=0
for threshold in /sys/class/power_supply/BAT*/charge_control_end_threshold; do
    [[ -e "$threshold" ]] || continue
    found=1
    printf '80\n' > "$threshold"
    printf 'Set %s to 80%%\n' "${threshold%/charge_control_end_threshold}"
done

if (( found == 0 )); then
    printf 'No battery exposing charge_control_end_threshold was found.\n' >&2
    exit 1
fi
